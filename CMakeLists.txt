cmake_minimum_required(VERSION 3.16)
project(epd7in5_v2_simulator LANGUAGES CXX)
include(cmake/SourceDependencies.cmake)
add_source_dependency(TARGET epd7in5_v2_api CAPABILITY display-api
  DESCRIPTION "Display interface" OVERRIDE DISPLAY_API_SOURCE_DIR)
if(NOT TARGET SDL2::SDL2)
  find_package(SDL2 2.0 CONFIG QUIET)
endif()
if(NOT TARGET SDL2::SDL2)
  message(FATAL_ERROR "SDL2 development files not found. Install SDL2 or set SDL2_DIR.")
endif()
add_library(epd7in5_v2_simulator src/epd7in5_V2.cpp src/runner.cpp)
target_compile_features(epd7in5_v2_simulator PUBLIC cxx_std_17)
target_include_directories(epd7in5_v2_simulator PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/include")
target_link_libraries(epd7in5_v2_simulator PUBLIC epd7in5_v2_api PRIVATE SDL2::SDL2)
if(CMAKE_SOURCE_DIR STREQUAL PROJECT_SOURCE_DIR)
  add_executable(display_preview examples/display_preview.cpp)
  target_link_libraries(display_preview PRIVATE epd7in5_v2_simulator)
  include(CTest)
  if(BUILD_TESTING)
    add_executable(simulator_checks tests/simulator_checks.cpp)
    target_link_libraries(simulator_checks PRIVATE epd7in5_v2_simulator SDL2::SDL2)
    function(add_simulator_check name width_mm height_mm ppi)
      add_test(NAME "${name}" COMMAND simulator_checks ${ARGN})
      set_tests_properties("${name}" PROPERTIES ENVIRONMENT
        "SDL_VIDEODRIVER=dummy;EPD_SIMULATOR_WIDTH_MM=${width_mm};EPD_SIMULATOR_HEIGHT_MM=${height_mm};EPD_SIMULATOR_PPI=${ppi}")
    endfunction()
    add_simulator_check(simulator_checks "" "" "")
    add_simulator_check(simulator_double_ppi "163.2" "97.92" "217.93221793221794" 1400 840)
    add_simulator_check(simulator_physical_dimensions "152.4" "91.44" "100" 600 360)
    add_simulator_check(simulator_invalid_settings "nan" "inf" "100junk")
    add_simulator_check(simulator_tiny_dimensions "0.000001" "0.000001" "100"
      --expect-init-failure)
    add_simulator_check(simulator_excessive_dimensions "1000000000000" "1000000000000" "100"
      --expect-init-failure)
  endif()
endif()
